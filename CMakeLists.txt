cmake_minimum_required(VERSION 3.16)
project(ArvoreAVL LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

add_library(avl_tree src/AVLTree.cpp)
target_include_directories(avl_tree PUBLIC include)
if(WIN32)
    add_executable(avl WIN32 src/gui.cpp)
    target_link_libraries(avl PRIVATE avl_tree comctl32 gdi32 user32)
    add_executable(avl_console src/main.cpp)
    target_link_libraries(avl_console PRIVATE avl_tree)
    set(AVL_CONSOLE_TARGET avl_console)
else()
    add_executable(avl src/main.cpp)
    target_link_libraries(avl PRIVATE avl_tree)
    set(AVL_CONSOLE_TARGET avl)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(avl_tests tests/avl_tests.cpp)
    target_link_libraries(avl_tests PRIVATE avl_tree)
    add_test(NAME avl_tests COMMAND avl_tests)
    add_executable(avl_memory_tests tests/memory_tests.cpp)
    target_link_libraries(avl_memory_tests PRIVATE avl_tree)
    add_test(NAME avl_memory_tests COMMAND avl_memory_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME avl_cli_tests
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py"
                         "$<TARGET_FILE:${AVL_CONSOLE_TARGET}>")
    endif()
endif()
